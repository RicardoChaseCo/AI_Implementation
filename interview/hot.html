1 两数之和

Two Sum · Easy

给定整数数组 nums 和目标 target。找出和为 target 的两个下标，每个下标只用一次。

例：nums = [2, 7, 11, 15]，target = 9 → [0, 1]

fun twoSum(nums: IntArray, target: Int): IntArray {
    val map = HashMap<Int, Int>() // 例子 [2,7,11,15]、目标 9：值对下标，先空着

    for (i in nums.indices) {
        val x = nums[i]
        val need = target - x

        val found = map[need]
        if (found != null) { // 先查：下标 1 的 7 要找 2
            return intArrayOf(found, i) // 查到就返回：2 在下标 0，得到 [0,1]
        }

        map[x] = i // 再写：下标 0 的 2 没找到 7，记下这个 2
    }

    return intArrayOf()
}

202 快乐数

Happy Number · Easy

把一个数换成各位数字的平方和，一直做下去。能变成 1 就是快乐数，数字开始循环就不是。

例：n = 19 → true。19 → 82 → 68 → 100 → 1

fun isHappy(n: Int): Boolean {
    val seen = HashSet<Int>() // 例子 19→82→68→100→1：走过的数先空着
    var x = n

    while (x != 1) { // 19、82、68、100 还要再算一步
        if (seen.contains(x)) { // 先查：19、82、68、100 都是第一次
            return false
        }
        seen.add(x) // 再写：先记 19，再记 82、68、100

        var sum = 0
        var y = x
        while (y > 0) {
            val digit = y % 10
            sum = sum + digit * digit // 19 的 1 和 9 平方相加，得到 82
            y = y / 10
        }
        x = sum // 换成 82，接着 68、100，再到 1
    }

    return true // 走到 1，19 是快乐数
}

242 有效的字母异位词

Valid Anagram · Easy

两个字符串是否由同样的字母、同样的次数组成。

例：s = "anagram"，t = "nagaram" → true。s = "rat"，t = "car" → false

fun isAnagram(s: String, t: String): Boolean {
    if (s.length != t.length) { // anagram 和 nagaram 都是 7 个字母
        return false
    }

    val count = HashMap<Char, Int>() // a 会到 3，n、g、r、m 各是 1

    for (ch in s) {
        val old = count[ch] ?: 0
        count[ch] = old + 1 // 先把 anagram 里的 a 加到 3
    }

    for (ch in t) {
        val old = count[ch] ?: 0
        if (old == 0) { // car 里的 c 次数是 0
            return false // rat 对不上 car，返回假
        }
        count[ch] = old - 1 // nagaram 把 a 从 3 减到 0
    }

    return true // 次数对完，anagram 和 nagaram 返回真
}

49 字母异位词分组

Group Anagrams · Medium

把互为异位词的字符串放进同一组。组的先后顺序、组内顺序都可以不同。

例：["eat","tea","tan","ate","nat","bat"] → 三组：eat/tea/ate，tan/nat，bat

fun groupAnagrams(strs: Array<String>): List<List<String>> {
    val map = HashMap<String, MutableList<String>>() // 例子 eat、tea、tan、ate、nat、bat

    for (s in strs) {
        val chars = s.toCharArray()
        chars.sort() // 算桶：eat、tea、ate 排成 aet，tan、nat 排成 ant
        val key = String(chars)

        var bucket = map[key]
        if (bucket == null) { // 没有桶就建：aet、ant、abt 各一次
            bucket = mutableListOf()
            map[key] = bucket
        }
        bucket.add(s) // 放进桶：tea、ate 跟 eat，nat 跟 tan，bat 单独
    }

    val answer = mutableListOf<List<String>>()
    for (bucket in map.values) {
        answer.add(bucket) // 收出三组：eat/tea/ate，tan/nat，bat
    }
    return answer
}

36 有效的数独

Valid Sudoku · Medium

只检查 9×9 数独里已经填上的数字。同一行、同一列、同一个 3×3 宫不能重复。'.' 是空格。

例：任一数字在它所在的行、列或宫里出现第二次 → false

fun isValidSudoku(board: Array<CharArray>): Boolean {
    val rows = Array(9) { HashSet<Char>() } // 9 行各放一个桶
    val cols = Array(9) { HashSet<Char>() } // 9 列各放一个桶
    val boxes = Array(9) { HashSet<Char>() } // 9 个宫各放一个桶

    for (r in 0 until 9) {
        for (c in 0 until 9) {
            val ch = board[r][c]
            if (ch == '.') { // '.' 是空格，跳过
                continue
            }

            val box = (r / 3) * 3 + (c / 3) // 算桶：三行三列归到同一宫
            if (rows[r].contains(ch) || cols[c].contains(ch) || boxes[box].contains(ch)) {
                return false // 桶里已有：行、列或宫里第二次出现，返回假
            }

            rows[r].add(ch)
            cols[c].add(ch)
            boxes[box].add(ch) // 放进桶：第一次同时记进行、列、宫
        }
    }

    return true
}

128 最长连续序列

Longest Consecutive Sequence · Medium

未排序的整数数组里，数字连续的最长一段有多长。看的是数值相邻，不是下标相邻。

例：nums = [100, 4, 200, 1, 3, 2] → 4，这段是 1, 2, 3, 4

fun longestConsecutive(nums: IntArray): Int {
    val set = HashSet<Int>() // 例子 [100,4,200,1,3,2]
    for (x in nums) {
        set.add(x) // 100、4、200、1、3、2 都收进来
    }

    var best = 0

    for (x in set) {
        if (set.contains(x - 1)) {
            continue // 4 左边有 3，3 左边有 2，2 左边有 1，都跳过
        }

        var y = x
        var length = 1
        while (set.contains(y + 1)) { // 从 1 往右，2、3、4 都还在
            y = y + 1
            length = length + 1 // 1 接到 2、3、4，长度记到 4
        }

        if (length > best) {
            best = length // 更长就换：1 到 4 比 100、200 的长度 1 更长
        }
    }

    return best
}

fun subarraySum(nums: IntArray, k: Int): Int {
    val map = HashMap<Int, Int>() // 例子 [1,1,1]，要凑的和是 2
    map[0] = 1 // 还没加数，前缀和 0 先记 1 次

    var prefix = 0
    var answer = 0

    for (x in nums) {
        prefix = prefix + x // 下标 0 的前缀是 1，下标 1 的前缀是 2，下标 2 的前缀是 3

        val need = prefix - k
        val found = map[need] ?: 0
        answer = answer + found // 前缀 2 配到 0，前缀 3 配到 1，两段都是 [1,1]

        val old = map[prefix] ?: 0
        map[prefix] = old + 1 // 配完再记：前缀 1、2、3 各 1 次
    }

    return answer
}

167 两数之和 II

Two Sum II · Medium

数组已经按非递减排好。找出和为 target 的两个下标，答案从 1 开始计数，且下标1 小于下标2。

例：numbers = [2, 7, 11, 15]，target = 9 → [1, 2]

fun twoSum(numbers: IntArray, target: Int): IntArray {
    var left = 0 // 例子 [2,7,11,15]、目标 9：左端下标 0 是 2
    var right = numbers.lastIndex // 右端下标 3 是 15

    while (left < right) {
        val sum = numbers[left] + numbers[right]
        if (sum == target) { // 正好：下标 0 的 2 加下标 1 的 7 等于 9
            // 正好就收：下标从 1 数，写成 [1,2]
            return intArrayOf(left + 1, right + 1)
        }
        if (sum < target) {
            left = left + 1
        } else {
            right = right - 1 // 偏大，右端往左收：2 加 15 得 17，15 收到 11 再收到 7
        }
    }

    return intArrayOf()
}

11 盛最多水的容器

Container With Most Water · Medium

每条垂线的高度给在 height 里。选两条线和 x 轴围出的容器，能装下的水最多是多少。

例：height = [1, 8, 6, 2, 5, 4, 8, 3, 7] → 49

fun maxArea(height: IntArray): Int {
    var left = 0 // 例子 [1,8,6,2,5,4,8,3,7]：左端下标 0 是 1
    var right = height.lastIndex // 右端下标 8 是 7
    var best = 0

    while (left < right) {
        val width = right - left
        // 下标 1 是 8、下标 8 是 7，高取矮的 7
        val high = if (height[left] < height[right]) height[left] else height[right]
        val area = width * high
        if (area > best) {
            best = area // 先是宽 8 高 1 的 8，再换成宽 7 高 7 的 49
        }

        // 先丢下标 0 的 1，面积 49 之后再丢下标 8 的 7
        if (height[left] <= height[right]) {
            left = left + 1 // 左端往右收：下标 0 的 1 比下标 8 的 7 矮
        } else {
            right = right - 1 // 右端往左收：下标 8 的 7 比下标 1 的 8 矮
        }
    }

    return best
}

125 验证回文串

Valid Palindrome · Easy

只看字母和数字，忽略大小写和其它字符，判断它是不是回文。

例：s = "A man, a plan, a canal: Panama" → true。s = "race a car" → false

fun isPalindrome(s: String): Boolean {
    var left = 0 // 例子：左端下标 0 是 A
    var right = s.lastIndex // 右端下标 29 是 a

    while (left < right) {
        // 左端往右收：下标 1 的空格让到下标 2 的 m
        while (left < right && !s[left].isLetterOrDigit()) {
            left = left + 1
        }
        // 右端往左收：下标 23 的空格、下标 22 的冒号，让到下标 21 的 l
        while (left < right && !s[right].isLetterOrDigit()) {
            right = right - 1
        }

        if (s[left].lowercaseChar() != s[right].lowercaseChar()) {
            return false // 例子 "race a car"：下标 3 的 e 对下标 5 的 a，返回假
        }
        left = left + 1 // 正好就收：下标 0 的 A 对上下标 29 的 a，左端再往右
        right = right - 1 // 右端往左收：从下标 29 的 a 收到下标 28 的 m
    }

    return true // 两端字母都对上，返回真
}

15 三数之和

3Sum · Medium

找出所有和为 0 的三元组。答案里不能有重复的三元组。

例：nums = [-1, 0, 1, 2, -1, -4] → [[-1, -1, 2], [-1, 0, 1]]

fun threeSum(nums: IntArray): List<List<Int>> {
    nums.sort() // 例子 [-1,0,1,2,-1,-4] 排成 [-4,-1,-1,0,1,2]
    val answer = mutableListOf<List<Int>>()

    for (i in nums.indices) {
        if (i > 0 && nums[i] == nums[i - 1]) {
            continue
        }

        var left = i + 1 // 第一次命中时，左端是下标 2 的 -1
        var right = nums.lastIndex // 右端下标 5 是 2
        val target = -nums[i] // 固定下标 1 的 -1，两数要凑 1

        while (left < right) {
            val sum = nums[left] + nums[right]
            if (sum == target) { // 正好：下标 2 的 -1 加下标 5 的 2 等于 1
                answer.add(listOf(nums[i], nums[left], nums[right])) // 正好就收：第一次得到 [-1,-1,2]
                left = left + 1
                right = right - 1
                while (left < right && nums[left] == nums[left - 1]) {
                    left = left + 1
                }
                while (left < right && nums[right] == nums[right + 1]) {
                    right = right - 1
                }
            } else if (sum < target) {

fun minWindow(s: String, t: String): String {

    for (ch in t) {
        val old = need[ch] ?: 0
        need[ch] = old + 1 // A、B、C 各要 1 次
    }

    val window = HashMap<Char, Int>()
    var have = 0
    val required = need.size
    var left = 0 // 左端从下标 0 的 A 出发，还缺 B、C
    var bestLeft = 0
    var bestLen = Int.MAX_VALUE

    for (right in s.indices) {
        val ch = s[right]
        val old = window[ch] ?: 0
        window[ch] = old + 1 // 右端加入：下标 5 的 C 补上缺的 C，下标 12 的 C 再补上
        // 下标 3 的 B 进来后还缺 C

            have = have + 1

            if (length < bestLen) {
                bestLen = length // 更短就换：先是下标 0 的 ADOBEC，再是下标 8 的 EBANC，最后下标 9 的 BANC
                bestLeft = left

                have = have - 1

            val leftOld = window[leftCh] ?: 0
            window[leftCh] = leftOld - 1
            left = left + 1
        }

        return ""

    return s.substring(bestLeft, bestLeft + bestLen) // 取下标 9 起的 BANC

fun maxSlidingWindow(nums: IntArray, k: Int): IntArray {
    val deque = ArrayDeque<Int>() // 例子 [1,3,-1,-3,5,3,6,7]，每格窗口 3 个数
    val answer = IntArray(nums.size - k + 1)

        // 下标 1 的 3 在滑到 [-1,-3,5] 时离开窗口

            deque.removeFirst()
        }
        // 队尾不大于新数就弹掉：1 碰到 3，-3 和 -1 碰到 5
        while (deque.isNotEmpty() && nums[deque.last()] <= nums[i]) {
            deque.removeLast()
        }
        deque.addLast(i) // 下标 1 的 3 留在队头，盖住 [1,3,-1]

            answer[write] = nums[deque.first()] // 依次写下 3、3、5、5、6、7
            write = write + 1
        }
    }

    return answer
}

        } else if (token == "-") {

            val left = stack.removeLast()
            stack.addLast(left - right)
        } else if (token == "*") {
            val right = stack.removeLast() // 乘号先弹出 3，再弹出 3
            val left = stack.removeLast()
            stack.addLast(left * right) // 压回 9
        } else if (token == "/") {
            val right = stack.removeLast()
            val left = stack.removeLast()
            stack.addLast(left / right)
        } else {
            stack.addLast(token.toInt()) // 2 和 1 入栈，加完再压上 3
        }
    }

    return stack.last()
}

        }

class MedianFinder() {
    val small = java.util.PriorityQueue<Int>(
        compareBy { value -> -value } // 较小一半用大顶堆，顶上先是1
    )
    val large = java.util.PriorityQueue<Int>(
        compareBy { value -> value } // 较大一半用小顶堆，2在顶，后来3在顶
    )

    fun addNum(num: Int) {
        small.add(num) // 1、2、3都先放进较小一半
        val moved = small.poll() // 再把较小一半的顶拨到较大一半
        if (moved != null) {
            large.add(moved)
        }

        if (large.size > small.size) { // 1先拨回；3进来后2也拨回
            val back = large.poll()
            if (back != null) {
                small.add(back)
            }
        }
    }

    fun findMedian(): Double { // 两个数时两边各一个
        if (small.size > large.size) { // 三个数时较小一半多，顶是2
            val mid = small.peek()
            if (mid == null) {
                return 0.0
            }
            return mid.toDouble()
        }

        val left = small.peek()
        val right = large.peek()
        if (left == null || right == null) {
            return 0.0
        }
        return (left.toDouble() + right.toDouble()) / 2.0 // 1和2平均得1.5
    }
}

}

        } else {
            rightMax[i] = height[i]

4 寻找两个正序数组的中位数

两个数组都已经升序。把它们合并后的中位数求出来。总数是偶数时，取中间两个数的平均值。

fun findMedianSortedArrays(nums1: IntArray, nums2: IntArray): Double {
    var a = nums1
    var b = nums2
    if (a.size > b.size) { // [1,3]更长，改为去切[2]
        val tmp = a
        a = b
        b = tmp
    }

    val m = a.size
    val n = b.size
    val half = (m + n + 1) / 2 // 左半固定要2个数
    var lo = 0
    var hi = m

    while (lo <= hi) {
        val cutA = lo + (hi - lo) / 2 // [1,2]第一刀把1留左、2留右
        val cutB = half - cutA // 配对刀把3留左、4留右

        var leftA = Int.MIN_VALUE
        if (cutA > 0) {
            leftA = a[cutA - 1]
        }
        var rightA = Int.MAX_VALUE
        if (cutA < m) {
            rightA = a[cutA]
        }

        var leftB = Int.MIN_VALUE
        if (cutB > 0) {
            leftB = b[cutB - 1]
        }
        var rightB = Int.MAX_VALUE
        if (cutB < n) {
            rightB = b[cutB]
        }

        if (leftA <= rightB && leftB <= rightA) { // 第一刀3比2大，这一次不进
            val leftPart: Int
            if (leftA > leftB) {
                leftPart = leftA
            } else {
                leftPart = leftB
            }

            if ((m + n) % 2 == 1) { // 刀在2之后、1与3之间，奇数取2
                return leftPart.toDouble()
            }

            val rightPart: Int
            if (rightA < rightB) {
                rightPart = rightA
            } else {
                rightPart = rightB
            }
            return (leftPart.toDouble() + rightPart.toDouble()) / 2.0 // 刀在2之后、3之前，四值1、2、3、4，2和3平均得2.5
        }

        if (leftA > rightB) { // 1和4比，1更小
            hi = cutA - 1
        } else {
            lo = cutA + 1 // 3比2大，丢掉靠左的切法
        }
    }

    return 0.0
}

            i = i + 1
        }
        if (i >= s.length) {
            break
        }

        while (i < s.length && s[i] != ' ') { // 从 t 把 the 读到空格前
            i = i + 1
        }
        stack.addLast(s.substring(start, i)) // 压入 the，接着 sky、is、blue
    }

    while (stack.isNotEmpty()) {
        if (sb.isNotEmpty()) {
            sb.append(' ') // blue 和 is 之间补一个空格
        }
        sb.append(stack.removeLast()) // 先出 blue，再出 is、sky、the
    }
    return sb.toString()
}

fun multiply(num1: String, num2: String): String {
    if (num1 == "0" || num2 == "0") {
        return "0"
    }

    val m = num1.length
    val n = num2.length
    val buf = IntArray(m + n) // 123 乘 456，六格先装没进位的积

    // 从个位的 3 乘 6 开始
    for (i in m - 1 downTo 0) {
        val a = num1[i] - '0'
        for (j in n - 1 downTo 0) {
            val b = num2[j] - '0'
            val pos = i + j + 1
            buf[pos] = buf[pos] + a * b // 3 乘 6 得到 18，加进末格
        }
    }

    for (k in buf.lastIndex downTo 1) {
        buf[k - 1] = buf[k - 1] + buf[k] / 10 // 18 的 1 进到前一格
        buf[k] = buf[k] % 10 // 末格写下 8，就是 56088 的末位
    }

    val sb = StringBuilder()
    var start = 0
    while (start < buf.lastIndex && buf[start] == 0) { // 丢掉最高位的 0
        start = start + 1
    }
    sb.append(buf[start]) // 从 5 开始写出 56088
    for (k in start + 1 until buf.size) {
        sb.append('0' + buf[k])
    }
    return sb.toString()
}

        return 1.0 // 指数折半到 0，得到 1
    }

    if (n == Int.MIN_VALUE) { // 最小整数指数是特例
        val half = myPow(x, n / 2) // 特例里先把指数折半
        return half * half
    }

    if (n < 0) {
        return 1.0 / myPow(x, -n) // 例子里指数是 -2，负指数要取倒数才得到 0.25
    }

    val half = myPow(x, n / 2) // 指数折半：10 到 5，再到 2、1
    val squared = half * half // 半份结果再平方
    // 10 是偶数就用平方；5 是奇数还要再乘 2.0
    if (n % 2 == 0) {
        return squared // 偶数 10 得到 1024
    }
    return squared * x // 奇数 5 再乘上 2.0
}

    while (i < s.length && s[i] == ' ') {  // "   -42" 先跳过三格空格
        i = i + 1
    }
    if (i >= s.length) {
        return 0
    }

    var sign = 1
    if (s[i] == '+' || s[i] == '-') {  // 空格后面是负号
        if (s[i] == '-') {
            sign = -1  // 记下负号，准备得到 -42
        }
        i = i + 1  // 跳过这个负号
    }

    var value = 0L  // 从 0 开始读数字
    while (i < s.length && s[i] in '0'..'9') {  // 42 读完；4193 遇到空格停止

            return Int.MAX_VALUE  // 正数越过上界就停在这里
        }
        if (sign == -1 && -value < Int.MIN_VALUE) {
            return Int.MIN_VALUE  // 负数越过下界就停在这里
        }
        i = i + 1
    }

    return (sign * value).toInt()  // 乘上负号得到 -42

LRU Cache · Medium

实现一个容量有限的缓存。读取和写入都要 O(1)。满了就淘汰最久没有被用到的键。

例：容量 2。put(1, 1)，put(2, 2)，get(1) → 1，put(3, 3) 之后 get(2) → -1

class LRUCache(private val capacity: Int) {
    private class Node(val key: Int, var value: Int) {
        var prev: Node? = null
        var next: Node? = null
    }

    private val map = HashMap<Int, Node>() // 记下键 1、2、3
    private val head = Node(0, 0)
    private val tail = Node(0, 0)

    init {
        head.next = tail
        tail.prev = head
    }

    fun get(key: Int): Int {
        val node = map[key] // 读取 1 能找到
        if (node == null) {
            return -1 // 键 2 被淘汰后再读，得到 -1
        }
        moveToFront(node) // 读取 1 之后，头是键 1
        return node.value // 返回 1
    }

    fun put(key: Int, value: Int) {
        val old = map[key]
        if (old != null) {
            old.value = value
            moveToFront(old)
            return
        }

        val node = Node(key, value)
        map[key] = node // 写入键 1，再写入 2 和 3
        insertFront(node) // 写入 2 时头是键 2，写入 3 时头是键 3

        if (map.size > capacity) { // 容量 2，写入 3 时超了
            val last = tail.prev
            if (last != null && last != head) {
                removeNode(last)
                map.remove(last.key) // 淘汰键 2
            }
        }
    }

    private fun moveToFront(node: Node) {
        removeNode(node)
        insertFront(node)
    }

    private fun removeNode(node: Node) {
        val prev = node.prev
        val next = node.next
        if (prev != null) {
            prev.next = next
        }
        if (next != null) {
            next.prev = prev
        }
    }

    private fun insertFront(node: Node) {
        val first = head.next
        node.prev = head
        node.next = first
        head.next = node
        if (first != null) {
            first.prev = node
        }
    }
}

460 LFU 缓存

LFU Cache · Hard

实现最不经常使用缓存。容量满了，删掉使用次数最少的键。次数相同，删掉最久没用的那个。

例：容量 2：put(1,1)、put(2,2)、get(1)、put(3,3) 后 get(2) → -1

class LFUCache(private val capacity: Int) {
    private class Node(val key: Int, var value: Int, var freq: Int) {
        var prev: Node? = null
        var next: Node? = null
    }

    private class Bucket {
        val head = Node(0, 0, 0)  // 同频次里靠近头的更新，键 2 曾排在键 1 前
        val tail = Node(0, 0, 0)

        init {
            head.next = tail
            tail.prev = head
        }

        fun isEmpty(): Boolean {
            return head.next == tail
        }

        fun addFront(node: Node) {
            val first = head.next
            node.prev = head
            node.next = first
            head.next = node
            if (first != null) {
                first.prev = node
            }
        }

        fun remove(node: Node) {
            val before = node.prev
            val after = node.next
            if (before != null) {
                before.next = after
            }
            if (after != null) {
                after.prev = before
            }
        }

        fun removeLast(): Node? {
            val last = tail.prev
            if (last == null || last == head) {
                return null
            }
            remove(last)
            return last
        }
    }

    private val nodes = HashMap<Int, Node>()  // 记下键 1、2、3
    private val buckets = HashMap<Int, Bucket>()  // 桶是频次
    private var minFreq = 0

    private fun bucket(freq: Int): Bucket {
        val found = buckets[freq]
        if (found != null) {
            return found
        }
        val created = Bucket()
        buckets[freq] = created
        return created
    }

    private fun touch(node: Node) {
        val oldBucket = bucket(node.freq)
        oldBucket.remove(node)
        if (oldBucket.isEmpty() && node.freq == minFreq) {  // 读取 1 后，频次 1 还留着键 2
            minFreq = minFreq + 1  // 桶空了才把 minFreq 加一
        }
        node.freq = node.freq + 1  // 键 1 的频次从 1 升到 2
        bucket(node.freq).addFront(node)  // 频次 2 的头是键 1
    }

    fun get(key: Int): Int {
        val node = nodes[key]
        if (node == null) {
            return -1  // 键 2 被删后再读，得到 -1
        }
        touch(node)
        return node.value
    }

    fun put(key: Int, value: Int) {
        if (capacity == 0) {
            return
        }
        val old = nodes[key]
        if (old != null) {
            old.value = value
            touch(old)
            return
        }
        if (nodes.size == capacity) {
            val least = bucket(minFreq).removeLast()  // minFreq 是 1，同频次里删最久的键 2
            if (least != null) {
                nodes.remove(least.key)
            }
        }
        val node = Node(key, value, 1)
        nodes[key] = node
        bucket(1).addFront(node)  // 写入 3 后，频次 1 的头是键 3
        minFreq = 1  // 新键频次是 1，minFreq 记成 1
    }
}

fun widthOfBinaryTree(root: TreeNode?): Int {
    if (root == null) {
        return 0  // 空树交还 0
    }

    val queue = ArrayDeque<Pair<TreeNode, Long>>()  // 根 1 编号 0，左 3，右 2
    queue.addLast(Pair(root, 0L))
    var best = 0

    while (queue.isNotEmpty()) {
        val size = queue.size
        val leftIndex = queue.first().second  // 第三层最左是 5
        var rightIndex = leftIndex
        var i = 0

        while (i < size) {
            val item = queue.removeFirst()
            val node = item.first
            val index = item.second - leftIndex  // 减去最左，空位才算进宽度
            rightIndex = item.second

            // 左孩子 3 底下是 5 和 3，右孩子 2 的右是 9
            val leftChild = node.left
            if (leftChild != null) {
                queue.addLast(Pair(leftChild, index * 2))
            }
            val rightChild = node.right
            if (rightChild != null) {
                queue.addLast(Pair(rightChild, index * 2 + 1))
            }
            i = i + 1
        }

        val width = (rightIndex - leftIndex + 1).toInt()  // 5、3、空、9 这一层宽度 4
        if (width > best) {
            best = width
        }
    }

    return best
}

fun isSubStructure(a: TreeNode?, b: TreeNode?): Boolean {
    if (a == null || b == null) {
        return false  // 空的一边交还假
    }

    fun match(x: TreeNode?, y: TreeNode?): Boolean {
        if (y == null) {
            return true  // B 走完即真，多出的 2 留下也算对
        }
        if (x == null || x.`val` != y.`val`) {
            return false  // 根 3 对上 B 的 4，值不同，这一处失败
        }
        return match(x.left, y.left) && match(x.right, y.right)  // 4 的左 1 对上 B 的 1
    }

    if (match(a, b)) {
        return true  // 换到结点 4 对上，交还真
    }
    // 根 3 对不上，就换左子 4 重新对
    return isSubStructure(a.left, b) || isSubStructure(a.right, b)
}

fun exist(board: Array<CharArray>, word: String): Boolean {
    val rows = board.size
    if (rows == 0) {
        return false
    }
    val cols = board[0].size
    val dirs = arrayOf(
        intArrayOf(1, 0),
        intArrayOf(-1, 0),
        intArrayOf(0, 1),
        intArrayOf(0, -1)
    )
    val seen = Array(rows) { BooleanArray(cols) }

    fun dfs(r: Int, c: Int, index: Int): Boolean { // 「ABCCED」的第一格是 (0,0) 的 A
        if (index == word.length) { // 走通时路径是 (0,0)A (0,1)B (0,2)C (1,2)C (2,2)E (2,1)D
            return true
        }
        if (r < 0 || c < 0 || r >= rows || c >= cols) {
            return false // 越出棋盘就停
        }
        if (seen[r][c]) {
            return false // 走过的不再踩，「ABCB」因此走不完
        }
        if (board[r][c] != word[index]) {
            return false // 向下的「S」不是「B」就停
        }

        seen[r][c] = true // 这一行追加，先占 (0,0)
        for (dir in dirs) { // 接着去 B、C、C、E、D
            val nr = r + dir[0]
            val nc = c + dir[1]
            if (dfs(nr, nc, index + 1)) {
                seen[r][c] = false
                return true
            }
        }
        seen[r][c] = false // 这一行撤销这一格
        return false
    }

    for (r in 0 until rows) {
        for (c in 0 until cols) {
            if (dfs(r, c, 0)) {
                return true
            }
        }
    }
    return false
}

    val cols = BooleanArray(n)
    val diag1 = BooleanArray(n * 2)
    val diag2 = BooleanArray(n * 2)
    val path = IntArray(n)

    fun dfs(row: Int) { // n=4，从第 0 行开始放
        if (row == n) { // 第一次记下时 path 是 [1,3,0,2]，即 .Q.. / ...Q / Q... / ..Q.
            val board = mutableListOf<String>()
            for (r in 0 until n) {
                val line = CharArray(n)
                for (k in 0 until n) {
                    line[k] = '.'
                }
                line[path[r]] = 'Q'
                board.add(String(line))
            }
            ans.add(board)
            return
        }

        for (c in 0 until n) {
            // 第 0 行试列 0 到 3；cols 记列已有皇后，diag1、diag2 记两条对角线，.Q.. 占列 1，diag1 下标 3，diag2 下标 1
            val d1 = row - c + n
            val d2 = row + c
            if (cols[c] || diag1[d1] || diag2[d2]) {
                continue // 列或对角线已被占就跳过
            }

            cols[c] = true // 这一行追加，第 0 行占列 1 和两条对角线

            diag2[d2] = true
            path[row] = c
            dfs(row + 1)
            cols[c] = false // 这一行撤销列和对角线
            diag1[d1] = false
            diag2[d2] = false
        }
    }

    dfs(0)
    return ans
}

    val path = mutableListOf<String>()

    fun palindrome(left: Int, right: Int): Boolean { // 「aab」里「a」「aa」「b」是回文

        var j = right
        while (i < j) {
            if (s[i] != s[j]) {
                return false
            }
            i = i + 1
            j = j - 1
        }
        return true
    }

        if (start == s.length) { // 第一次记下时 path 是 [「a」,「a」,「b」]，还有 [「aa」,「b」]

            return

            path.add(s.substring(start, end + 1)) // 这一行追加，先放进「a」
            dfs(end + 1) // 再切出第二个「a」，然后「b」
            path.removeAt(path.lastIndex) // 这一行撤销，丢掉刚切的一段
        }
    }

    dfs(0)
    return ans
}

    val graph = Array(numCourses) { mutableListOf<Int>() }
    val indegree = IntArray(numCourses)

    for (pair in prerequisites) {
        val course = pair[0]
        val pre = pair[1]
        graph[pre].add(course) // 建边：pre → course，[1,0] 是 0 → 1
        indegree[course] = indegree[course] + 1 // [1,0] 使课 1 入度为 1；再加上 [0,1] 两边都是 1

    val queue = ArrayDeque<Int>()

        if (indegree[i] == 0) {
            queue.addLast(i) // 零入度入队，[1,0] 时课 0 入队
        }
    }

    var seen = 0
    while (queue.isNotEmpty()) {
        val cur = queue.removeFirst()
        seen = seen + 1
        for (next in graph[cur]) {
            indegree[next] = indegree[next] - 1 // 弹出后后继入度减一，课 1 变成 0
            if (indegree[next] == 0) {
                queue.addLast(next) // 课 1 减到 0 再入队
            }
        }
    }

    return seen == numCourses // 弹出的门数等于总门数，没有环

fun hasCircularDependency(n: Int, prerequisites: Array<IntArray>): Boolean {
    val graph = Array(n) { mutableListOf<Int>() }  // 两个模块 0 和 1
    val indegree = IntArray(n)  // 先看依赖 [1,0]

    for (pair in prerequisites) {
        val module = pair[0]
        val depend = pair[1]
        graph[depend].add(module)  // 建边：pre → course，[1,0] 是 0 → 1，加上 [0,1] 是 1 → 0
        indegree[module] = indegree[module] + 1
    }

    val queue = ArrayDeque<Int>()
    for (i in 0 until n) {
        if (indegree[i] == 0) {  // 零入度入队，[1,0] 时模块 0 入队
            queue.addLast(i)
        }
    }

    var seen = 0
    while (queue.isNotEmpty()) {
        val cur = queue.removeFirst()
        seen = seen + 1
        for (next in graph[cur]) {
            indegree[next] = indegree[next] - 1  // 弹出后后继入度减一，模块 1 变成 0
            if (indegree[next] == 0) {
                queue.addLast(next)
            }
        }
    }

    return seen != n  // 看每个点是不是都走过；[1,0] 都走过，加上 [0,1] 后不是每个点都走过
}

class Node(var `val`: Int) {
    var neighbors: ArrayList<Node?> = ArrayList()
}

fun cloneGraph(node: Node?): Node? {
    if (node == null) {
        return null
    }

    val copies = HashMap<Node, Node>() // 旧的 1 对应新的 1
    val queue = ArrayDeque<Node>()
    val first = Node(node.`val`)
    copies[node] = first
    queue.addLast(node) // 从 1 入队

    while (queue.isNotEmpty()) {
        val cur = queue.removeFirst() // 先弹出 1
        val copy = copies[cur]
        if (copy == null) {
            continue
        }

        for (next in cur.neighbors) {
            if (next == null) {
                continue
            }

            var nextCopy = copies[next]
            if (nextCopy == null) {
                nextCopy = Node(next.`val`) // 第一次见到 2 和 4 就建新结点
                copies[next] = nextCopy
                queue.addLast(next) // 2 和 4 入队，随后才到 3
            }
            copy.neighbors.add(nextCopy) // 新的 1 连上新的 2 和 4
        }
    }

    return first
}

fun ladderLength(beginWord: String, endWord: String, wordList: List<String>): Int {
    val words = HashSet<String>()
    for (word in wordList) {
        words.add(word)
    }
    if (!words.contains(endWord)) {
        return 0
    }

    val dist = HashMap<String, Int>() // 「hit」的长度记成 1
    dist[beginWord] = 1
    val queue = ArrayDeque<String>()
    queue.addLast(beginWord)

    while (queue.isNotEmpty()) {
        val cur = queue.removeFirst() // 先弹出「hit」
        val steps = dist[cur]
        if (steps == null) {
            continue
        }
        if (cur == endWord) {
            return steps // 到「cog」时长度是 5
        }

        val chars = cur.toCharArray()
        for (i in chars.indices) {
            val saved = chars[i]
            var ch = 'a'
            while (ch <= 'z') { // 把「hit」里的一个字母换掉
                if (ch != saved) {
                    chars[i] = ch
                    val next = String(chars)
                    if (words.contains(next) && !dist.containsKey(next)) { // 「hit」第一次变成「hot」
                        dist[next] = steps + 1 // 「hot」的长度是 2
                        queue.addLast(next) // 「hot」入队
                    }
                }
                ch = ch + 1
            }
            chars[i] = saved
        }
    }

    return 0
}

    val n = grid[0].size

    val m = matrix.size
    val n = matrix[0].size
    val dp = Array(m) { IntArray(n) } // dp[i][j]：以 (i,j) 为右下角的正方形边长
    var best = 0

    for (i in 0 until m) { // 四行从上往下填
        for (j in 0 until n) {
            if (matrix[i][j] == '1') {
                if (i == 0 || j == 0) {
                    dp[i][j] = 1 // 边上的 1 边长写成 1，如 (0,0)、(0,2)
                } else {
                    var side = dp[i - 1][j]
                    if (dp[i][j - 1] < side) {
                        side = dp[i][j - 1]
                    }
                    if (dp[i - 1][j - 1] < side) {
                        side = dp[i - 1][j - 1]
                    }
                    dp[i][j] = side + 1 // (2,3)：左、上、左上都是 1，加 1 得 2

                val area = dp[i][j] * dp[i][j]
                if (area > best) {
                    best = area // 边长 2，面积写成 4
                }

        }
    }

}

fun longestCommonSubsequence(text1: String, text2: String): Int {
    val m = text1.length
    val n = text2.length
    val dp = Array(m + 1) { IntArray(n + 1) } // dp[i][j]：前 i 个对前 j 个的最长公共长度

    for (i in 1..m) {
        for (j in 1..n) { // 比较两个串当前的结尾
            if (text1[i - 1] == text2[j - 1]) {
                dp[i][j] = dp[i - 1][j - 1] + 1 // a 对 a 得 dp[1][1]=1，e 对 e 得 dp[5][3]=3
            } else {
                // 不相等就取上一格或左一格里较大的
                val up = dp[i - 1][j]
                val left = dp[i][j - 1]
                if (up > left) {
                    dp[i][j] = up
                } else {
                    dp[i][j] = left // b 对 c，上格和左格都是 1，dp[2][2]=1
                }
            }
        }
    }

    return dp[m][n] // abcde 对 ace，答案格 dp[5][3] 是 3
}

Longest Palindromic Substring · Medium

找出字符串里最长的一段回文子串。子串必须连续，同样长时返回任意一段即可。

例：s = "babad" → "bab"。s = "cbbd" → "bb"

fun longestPalindrome(s: String): String {
    var start = 0
    var best = 0

    fun expand(left0: Int, right0: Int) {
        var left = left0
        var right = right0

        // 左右字符相同就再向外扩一格
        while (left >= 0 && right < s.length && s[left] == s[right]) {
            val len = right - left + 1
            if (len > best) {
                best = len // babad 长度记到 3，cbbd 长度记到 2
                start = left
            }
            left = left - 1
            right = right + 1
        }
    }

    for (i in s.indices) { // 中心放在下标 i
        expand(i, i) // 奇数中心是下标 1 的 a，向左 1 步到 b、向右 1 步到 b，得到 bab
        expand(i, i + 1) // 偶数中心夹在 cbbd 下标 1 和 2 的两个 b 之间，扩出 bb
    }

    return s.substring(start, start + best)
}

    val n = word2.length

    }
    for (j in 0..n) {
        dp[0][j] = j // 空串插成 ros，dp[0][3]=3
    }

        for (j in 1..n) { // 比较两个词的当前结尾

                dp[i][j] = dp[i - 1][j - 1] // o 对上 o，dp[2][2] 沿用 1

                if (dp[i][j - 1] < best) {
                    best = dp[i][j - 1] // 插入 ros 的当前字符

                if (dp[i - 1][j - 1] < best) {

        }

fun isInterleave(s1: String, s2: String, s3: String): Boolean {
    val m = s1.length
    val n = s2.length
    if (m + n != s3.length) {
        return false
    }

    val dp = Array(m + 1) { BooleanArray(n + 1) } // dp[i][j]：s1 前 i 个、s2 前 j 个能否拼出 s3 的前 i+j 个
    dp[0][0] = true // 空串对空串，为真

        for (j in 0..n) { // 逐格看 s3 的下一位
            if (i == 0 && j == 0) {
                continue
            }

            if (i > 0 && dp[i - 1][j] && s1[i - 1] == s3[i + j - 1]) {
                dp[i][j] = true // 从上面来：dp[3][1]，s1 的 b 对上 s3 的 b

                dp[i][j] = true // 从左边来：dp[2][1]，s2 的 d 对上 s3 的 d
            }
        }
    }

    return dp[m][n] // dp[5][5]：aadbbcbcac 为真，aadbbbaccc 为假
}

fun quickSort(nums: IntArray): IntArray {
    fun swap(i: Int, j: Int) {
        val tmp = nums[i]
        nums[i] = nums[j]
        nums[j] = tmp
    }

    fun partition(lo: Int, hi: Int): Int {
        val pivot = nums[hi]  // 第一趟基准是 1
        var store = lo  // 从下标 0 收不大于 1 的
        var j = lo
        while (j < hi) {
            if (nums[j] <= pivot) {  // 5、2、3 都比基准 1 大
                swap(store, j)  // 比 1 小才会换到左边
                store = store + 1
            }
            j = j + 1
        }
        swap(store, hi)  // 5 和基准 1 对调，得到 [1,2,3,5]
        return store  // 基准 1 落在下标 0
    }

    fun sort(lo: Int, hi: Int) {
        if (lo >= hi) {
            return
        }
        val mid = partition(lo, hi)
        sort(lo, mid - 1)  // 基准 1 左边没有数
        sort(mid + 1, hi)  // 右边剩下 [2,3,5]
    }

    if (nums.isNotEmpty()) {
        sort(0, nums.lastIndex)
    }
    return nums
}

fun heapSort(nums: IntArray): IntArray {
    fun swap(i: Int, j: Int) {
        val tmp = nums[i]
        nums[i] = nums[j]
        nums[j] = tmp
    }

    fun siftDown(size: Int, start: Int) {
        var i = start
        while (true) {
            val left = i * 2 + 1
            val right = left + 1
            var largest = i  // 建堆先看下标 1 的 2
            if (left < size && nums[left] > nums[largest]) {  // 孩子 1 不比 2 大
                largest = left
            }
            if (right < size && nums[right] > nums[largest]) {  // 5 的右孩子 3 并不更大
                largest = right
            }
            if (largest == i) {  // 2 不用换，根上的 5 也不用换
                break
            }
            swap(i, largest)  // 1 和 3 对调
            i = largest
        }
    }

    val n = nums.size
    var i = n / 2 - 1
    while (i >= 0) {  // 从 2 建到 5，数组仍是 [5,2,3,1]
        siftDown(n, i)
        i = i - 1
    }

    var end = n - 1
    while (end > 0) {
        swap(0, end)  // 5 换到末尾，得到 [1,2,3,5]
        siftDown(end, 0)  // 1 下沉后得到 [3,2,1,5]
        end = end - 1
    }
    return nums
}

        sort(mid + 1, hi)

        var j = mid + 1

                temp[k] = nums[i]
                i = i + 1
            } else {
                temp[k] = nums[j]  // 先放右段的 2
                j = j + 1
            }
            k = k + 1
        }
        while (i <= mid) {

            k = k + 1
        }
        while (j <= hi) {
            temp[k] = nums[j]
            j = j + 1
            k = k + 1
        }
        for (t in lo..hi) {
            nums[t] = temp[t]  // 写回成 [2,5]
        }

fun reversePairs(nums: IntArray): Int {
    if (nums.isEmpty()) {
        return 0
    }
    val temp = IntArray(nums.size)  // 临时数组，先装 5 和 7

    fun sort(lo: Int, hi: Int): Int {
        if (lo >= hi) {
            return 0
        }
        val mid = lo + (hi - lo) / 2  // 切开，最先合并左段 [7]、右段 [5]
        // 左右先递归，第一段里有 7 和 5
        var count = sort(lo, mid) + sort(mid + 1, hi)

        var i = lo
        var j = mid + 1
        var k = lo
        while (i <= mid && j <= hi) {
            if (nums[i] <= nums[j]) {  // 左段 7 比右段 5 大
                temp[k] = nums[i]
                i = i + 1
            } else {
                temp[k] = nums[j]  // 先放右段的 5
                count = count + (mid - i + 1)  // 7 和 5 是逆序，加上 1
                j = j + 1
            }
            k = k + 1
        }
        while (i <= mid) {
            temp[k] = nums[i]  // 剩下左段的 7
            i = i + 1
            k = k + 1
        }
        while (j <= hi) {
            temp[k] = nums[j]
            j = j + 1
            k = k + 1
        }
        for (t in lo..hi) {
            nums[t] = temp[t]  // 写回成 [5,7]
        }
        return count
    }

    return sort(0, nums.lastIndex)  // 五对合计得到 5
}
