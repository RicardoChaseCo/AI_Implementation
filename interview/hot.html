1 两数之和

Two Sum · Easy

fun twoSum(nums: IntArray, target: Int): IntArray {
    val map = HashMap<Int, Int>()

    for (i in nums.indices) {
        val x = nums[i]
        val need = target - x

        val found = map[need]
        if (found != null) {
            return intArrayOf(found, i)
        }

        map[x] = i
    }

    return intArrayOf()
}

217 存在重复元素

Contains Duplicate · Easy

fun containsDuplicate(nums: IntArray): Boolean {
    val seen = HashSet<Int>()

    for (i in nums.indices) {
        val x = nums[i]

        if (seen.contains(x)) {
            return true
        }

        seen.add(x)
    }

    return false
}

Contains Duplicate II · Easy

fun containsNearbyDuplicate(nums: IntArray, k: Int): Boolean {
    val map = HashMap<Int, Int>()

    for (i in nums.indices) {
        val x = nums[i]
        val found = map[x]

        if (found != null && i - found <= k) {
            return true
        }

        map[x] = i
    }

    return false
}

202 快乐数

Happy Number · Easy

fun isHappy(n: Int): Boolean {
    val seen = HashSet<Int>()
    var x = n

    while (x != 1) {
        if (seen.contains(x)) {
            return false
        }
        seen.add(x)

        var sum = 0
        var y = x
        while (y > 0) {
            val digit = y % 10
            sum = sum + digit * digit
            y = y / 10
        }
        x = sum
    }

    return true
}

205 同构字符串

Isomorphic Strings · Easy

fun isIsomorphic(s: String, t: String): Boolean {
    if (s.length != t.length) {
        return false
    }

    val sToT = HashMap<Char, Char>()
    val tToS = HashMap<Char, Char>()

    for (i in s.indices) {
        val a = s[i]
        val b = t[i]

        val aMapped = sToT[a]
        if (aMapped != null && aMapped != b) {
            return false
        }

        val bMapped = tToS[b]
        if (bMapped != null && bMapped != a) {
            return false
        }

        sToT[a] = b
        tToS[b] = a
    }

    return true
}

fun wordPattern(pattern: String, s: String): Boolean {
    val words = s.split(" ")
    if (pattern.length != words.size) {
        return false
    }

    val pToW = HashMap<Char, String>()
    val wToP = HashMap<String, Char>()

    for (i in pattern.indices) {
        val a = pattern[i]
        val b = words[i]

        val aMapped = pToW[a]
        if (aMapped != null && aMapped != b) {
            return false
        }

        val bMapped = wToP[b]
        if (bMapped != null && bMapped != a) {
            return false
        }

        pToW[a] = b
        wToP[b] = a
    }

    return true
}

49 字母异位词分组

Group Anagrams · Medium

fun groupAnagrams(strs: Array<String>): List<List<String>> {
    val map = HashMap<String, MutableList<String>>()

    for (s in strs) {
        val chars = s.toCharArray()
        chars.sort()
        val key = String(chars)

        var bucket = map[key]
        if (bucket == null) {
            bucket = mutableListOf()
            map[key] = bucket
        }
        bucket.add(s)
    }

    val answer = mutableListOf<List<String>>()
    for (bucket in map.values) {
        answer.add(bucket)
    }
    return answer
}

36 有效的数独

Valid Sudoku · Medium

fun isValidSudoku(board: Array<CharArray>): Boolean {
    val rows = Array(9) { HashSet<Char>() }
    val cols = Array(9) { HashSet<Char>() }
    val boxes = Array(9) { HashSet<Char>() }

    for (r in 0 until 9) {
        for (c in 0 until 9) {
            val ch = board[r][c]
            if (ch == '.') {
                continue
            }

            val box = (r / 3) * 3 + (c / 3)
            if (rows[r].contains(ch) || cols[c].contains(ch) || boxes[box].contains(ch)) {
                return false
            }

            rows[r].add(ch)
            cols[c].add(ch)
            boxes[box].add(ch)
        }
    }

    return true
}

128 最长连续序列

Longest Consecutive Sequence · Medium

fun longestConsecutive(nums: IntArray): Int {
    val set = HashSet<Int>()
    for (x in nums) {
        set.add(x)
    }

    var best = 0

    for (x in set) {
        if (set.contains(x - 1)) {
            continue
        }

        var y = x
        var length = 1
        while (set.contains(y + 1)) {
            y = y + 1
            length = length + 1
        }

        if (length > best) {
            best = length
        }
    }

    return best
}

846 一手顺子

Hand of Straights · Medium

fun isNStraightHand(hand: IntArray, groupSize: Int): Boolean {
    if (hand.size % groupSize != 0) {
        return false
    }

    val count = HashMap<Int, Int>()
    for (card in hand) {
        val old = count[card] ?: 0
        count[card] = old + 1
    }

    val starts = count.keys.sorted()

    for (start in starts) {
        val need = count[start] ?: 0
        if (need == 0) {
            continue
        }

        for (offset in 0 until groupSize) {
            val card = start + offset
            val have = count[card] ?: 0
            if (have < need) {
                return false
            }
            count[card] = have - need
        }
    }

    return true
}

560 和为 K 的子数组

Subarray Sum Equals K · Medium

fun subarraySum(nums: IntArray, k: Int): Int {
    val map = HashMap<Int, Int>()
    map[0] = 1

    var prefix = 0
    var answer = 0

    for (x in nums) {
        prefix = prefix + x

        val need = prefix - k
        val found = map[need] ?: 0
        answer = answer + found

        val old = map[prefix] ?: 0
        map[prefix] = old + 1
    }

    return answer
}

437 路径总和 III

Path Sum III · Medium

class TreeNode(var `val`: Int) {
    var left: TreeNode? = null
    var right: TreeNode? = null
}

fun pathSum(root: TreeNode?, targetSum: Int): Int {
    val map = HashMap<Long, Int>()
    map[0L] = 1

    fun dfs(node: TreeNode?, prefix: Long): Int {
        if (node == null) {
            return 0
        }

        val current = prefix + node.`val`.toLong()
        val need = current - targetSum.toLong()
        var answer = map[need] ?: 0

        val old = map[current] ?: 0
        map[current] = old + 1

        answer = answer + dfs(node.left, current)
        answer = answer + dfs(node.right, current)

        map[current] = old
        return answer
    }

    return dfs(root, 0L)
}

13 罗马数字转整数

Roman to Integer · Easy

fun romanToInt(s: String): Int {
    val value = HashMap<Char, Int>()
    value['I'] = 1
    value['V'] = 5
    value['X'] = 10
    value['L'] = 50
    value['C'] = 100
    value['D'] = 500
    value['M'] = 1000

    var answer = 0

    for (i in s.indices) {
        val current = value.getValue(s[i])
        val next = if (i + 1 < s.length) value.getValue(s[i + 1]) else 0

        if (current < next) {
            answer = answer - current
        } else {
            answer = answer + current
        }
    }

    return answer
}

2013 检测正方形

Detect Squares · Medium

class DetectSquares() {
    private val points = HashMap<Int, HashMap<Int, Int>>()

    fun add(point: IntArray) {
        val x = point[0]
        val y = point[1]

        var row = points[x]
        if (row == null) {
            row = HashMap()
            points[x] = row
        }

        val old = row[y] ?: 0
        row[y] = old + 1
    }

    fun count(point: IntArray): Int {
        val x = point[0]
        val y = point[1]
        val row = points[x]
        if (row == null) {
            return 0
        }

        var answer = 0

        for ((py, times) in row) {
            val side = y - py
            if (side == 0) {
                continue
            }

            answer = answer + square(x + side, y, py, times)
            answer = answer + square(x - side, y, py, times)
        }

        return answer
    }

    private fun square(otherX: Int, y: Int, py: Int, times: Int): Int {
        val other = points[otherX]
        if (other == null) {
            return 0
        }
        val cornerY = other[y] ?: 0
        val cornerPy = other[py] ?: 0
        return times * cornerY * cornerPy
    }
}
